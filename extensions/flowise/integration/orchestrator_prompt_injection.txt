        print("\n" + enhanced_prompt)

1. Load Flowise architecture patterns:

```python
import sys
from pathlib import Path

cf_base = Path(__file__).parent.parent
flowise_ext = cf_base / 'extensions' / 'flowise'

if flowise_ext.exists():
    sys.path.insert(0, str(flowise_ext.parent))
    from flowise import extensions_loader

    # Get flow type from Scout findings
    flow_type = project_indicators.get('flowise_flow_type', 'unknown')

    # Load architecture enhancement
    enhancement = extensions_loader.get_extension_prompt('flowise', 'architect')

    if enhancement:
        # Format with flow type
        enhanced_prompt = enhancement.format(flow_type=flow_type)

        # Append to Architect guidance
        print("\n" + enhanced_prompt)

    # Also load pattern library
    patterns = extensions_loader.load_extension_patterns('flowise')

    if patterns:
        print("\n**FLOWISE PATTERN LIBRARY LOADED**")
        print(f"Available patterns: {len(patterns.get('patterns', []))}")

        # Filter patterns applicable to detected flow type
        applicable = [
            p for p in patterns.get('patterns', [])
            if flow_type in p.get('applies_to', [])
        ]

        if applicable:
            print(f"\n**PATTERNS FOR {flow_type.upper()} FLOWS:**")
            for pattern in applicable:
                print(f"\n- **{pattern['pattern_id']}**: {pattern['description']}")
                print(f"  Best practices: {', '.join(pattern.get('best_practices', [])[:3])}")
```

This provides Architect with proven Flowise patterns and best practices.
